ضع هذا الكود كاملاً في Apps Script بدلاً من الكود السابق، ثم احفظ وانشر إصداراً جديداً من Web app.

const SPREADSHEET_ID = '195E4GIjM5CQT3EYLXL1cpfEb4xG67n8UChteZafn41k';

function getSheet() {
  return SpreadsheetApp.openById(SPREADSHEET_ID).getSheets()[0];
}

function json(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  const rows = getSheet().getDataRange().getValues();
  const offers = rows.slice(1)
    .filter(row => String(row[6]).trim() === 'منشور')
    .map(row => ({
      id: row[0], title: row[1], price: row[2], date: row[3],
      details: row[4], image: row[5], status: row[6],
      category: row[7] || 'عرض سياحي'
    }));
  return json({ success: true, offers: offers });
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents || '{}');
    const secret = PropertiesService.getScriptProperties().getProperty('ADMIN_PASSWORD');
    if (!secret || data.password !== secret) throw new Error('كلمة المرور غير صحيحة');
    if (data.action === 'auth') return json({ success: true });

    const sheet = getSheet();
    const rows = sheet.getDataRange().getValues();
    if (data.action === 'add') {
      const id = Utilities.getUuid();
      sheet.appendRow([id, data.title || '', data.price || '', data.date || '',
        data.details || '', data.image || '', 'منشور', data.category || 'عرض سياحي']);
      return json({ success: true, id: id });
    }
    const index = rows.findIndex((row, i) => i > 0 && String(row[0]) === String(data.id));
    if (index < 1) throw new Error('العرض غير موجود');
    if (data.action === 'delete') {
      sheet.deleteRow(index + 1);
    } else if (data.action === 'update') {
      sheet.getRange(index + 1, 2, 1, 7).setValues([[
        data.title || '', data.price || '', data.date || '', data.details || '',
        data.image || '', 'منشور', data.category || 'عرض سياحي'
      ]]);
    } else {
      throw new Error('عملية غير معروفة');
    }
    return json({ success: true });
  } catch (err) {
    return json({ success: false, error: err.message });
  }
}

تأكد أن Script Properties يحتوي على:
ADMIN_PASSWORD = كلمة المرور التي اخترتها بنفسك.
